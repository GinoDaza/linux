fdisk -l

mount /dev/sdX2 /mnt

mount /dev/sdX1 /mnt/boot/efi

reflector --country "Brazil,Chile" --protocol https --sort rate --latest 10 --save /etc/pacman.d/mirrorlist

pacstrap -K /mnt base linux linux-firmware

genfstab -U /mnt >> /mnt/etc/fstab

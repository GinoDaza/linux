# Set Keyboard and Font
loadkeys la-latin1
setfont ter-122b

# Connect to Internet (Laptop)
iwctl
  station wlan0 connect (nombre del wifi)

# Partition and Format Disks
fdisk -l
fdisk /dev/...
mkfs.ext4 /dev/root_partition

# Mount Partitions
mount /dev/sdX2 /mnt
mkdir -p /mnt/boot/efi
mount /dev/sdX1 /mnt/boot/efi

# Install Base System
pacstrap -K /mnt base linux linux-firmware

# Generate Filesystem Table
genfstab -U /mnt >> /mnt/etc/fstab

# Enter Chroot
arch-chroot /mnt

# Set Time and Locale
ln -sf /usr/share/zoneinfo/Region/City /etc/localtime
hwclock --systohc

# Localization
nano /etc/locale.gen
locale-gen

# Config TTY
pacman -S terminus-font
echo "LANG=en_US.UTF-8" > /etc/locale.conf
echo -e "KEYMAP=la-latin1\nFONT=ter-122b" > /etc/vconsole.conf

# Network Configuration
echo "yourhostname" > /etc/hostname

# Install Bootloader
pacman -S grub efibootmgr os-prober
grub-install --target=x86_64-efi --efi-directory=/boot/efi --bootloader-id=GRUB
nano /etc/default/grub
grub-mkconfig -o /boot/grub/grub.cfg

# Create User and Configure Sudo
useradd -m -G wheel -s /bin/bash tu_usuario
passwd tu_usuario
pacman -S vi
visudo
# Uncomment: %wheel ALL=(ALL) ALL

# Install Drivers and Firmware (intel)
pacman -S sof-firmware mesa xf86-video-intel

# Install and Configure NetworkManager
pacman -S networkmanager
systemctl enable NetworkManager
nmtui
